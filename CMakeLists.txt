cmake_minimum_required(VERSION 3.24)
# The native Linux port is opt-in (README.md, section Linux). Without the option, this file is the
# Windows build, unchanged.
option(X4VR_LINUX "Build the native Linux port instead of the Windows mod" OFF)
if(X4VR_LINUX)
  project(x4vr_linux LANGUAGES C CXX)
  include(CTest)
  add_subdirectory(linux)
  return()
endif()
if(NOT CMAKE_HOST_WIN32)
  message(FATAL_ERROR "This integration targets Windows x64 X4. For the native Linux port, configure with -DX4VR_LINUX=ON.")
endif()
project(x4_openvr LANGUAGES C CXX ASM_MASM)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This integration targets Windows x64 X4.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Prebuilt downloads must run without a matching Visual C++ redistributable: the VC and C++
# runtimes are linked in, the Universal CRT (part of Windows 10 and 11) stays shared, so memory
# passed between our DLLs still comes from one heap.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_link_options("$<$<NOT:$<CONFIG:Debug>>:/NODEFAULTLIB:libucrt.lib>" "$<$<NOT:$<CONFIG:Debug>>:ucrt.lib>")
set(VULKAN_HEADERS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/external/Vulkan-Headers" CACHE PATH "Khronos Vulkan headers")
set(OPENVR_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/external/openvr" CACHE PATH "Valve OpenVR SDK checkout")
set(MINHOOK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/external/minhook" CACHE PATH "Pinned MinHook checkout")
set(OPENXR_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/external/OpenXR-SDK" CACHE PATH "Khronos OpenXR SDK checkout")
if(NOT EXISTS "${OPENVR_ROOT}/headers/openvr.h" OR NOT EXISTS "${OPENXR_ROOT}/include/openxr/openxr.h")
  message(FATAL_ERROR "Fetch the pinned SDKs using scripts/bootstrap.ps1 first.")
endif()
# OpenXR loader built from source as a static library, so no extra DLL ships. The SDK links a
# static loader against the DLL CRT; switch it to our static CRT.
set(DYNAMIC_LOADER OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_API_LAYERS OFF CACHE BOOL "" FORCE)
set(BUILD_CONFORMANCE_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${OPENXR_ROOT}" external/OpenXR-SDK EXCLUDE_FROM_ALL)
set_target_properties(openxr_loader PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT EXISTS "${MINHOOK_ROOT}/include/MinHook.h")
  message(FATAL_ERROR "Fetch pinned MinHook using scripts/bootstrap.ps1 first.")
endif()
# Kept private and static: do not share hook ownership with another module.
# This library is NOT linked into the Vulkan observation layer. The separate
# native module below is loaded only by the explicit startup-only PoseHook mode.
add_library(x4_minhook STATIC "${MINHOOK_ROOT}/src/buffer.c"
  "${MINHOOK_ROOT}/src/hook.c" "${MINHOOK_ROOT}/src/trampoline.c"
  "${MINHOOK_ROOT}/src/hde/hde64.c")
target_include_directories(x4_minhook PUBLIC "${MINHOOK_ROOT}/include")
add_library(x4_pose_detour STATIC src/pose_detour.cpp)
target_include_directories(x4_pose_detour PUBLIC include)
target_compile_definitions(x4_pose_detour PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(x4_pose_detour PRIVATE /W4 /WX /permissive-)
target_link_libraries(x4_pose_detour PRIVATE x4_minhook)
add_library(x4_copy_call_hook STATIC src/copy_call_hook.cpp)
target_include_directories(x4_copy_call_hook PUBLIC include)
target_compile_definitions(x4_copy_call_hook PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(x4_copy_call_hook PRIVATE /W4 /WX /permissive-)

target_include_directories(x4vr_native_pose PRIVATE src)

target_compile_definitions(x4vr_native_pose PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(x4vr_native_pose PRIVATE /W4 /WX /permissive-)
add_custom_command(TARGET x4vr_native_pose POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
  "${MINHOOK_ROOT}/LICENSE.txt" "$<TARGET_FILE_DIR:x4vr_native_pose>/MinHook-LICENSE.txt")

  src/x4_camera.cpp src/head_look.cpp)
